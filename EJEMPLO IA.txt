<script>
  const inputPeticion = document.getElementById('peticion');
  const inputPregunta = document.getElementById('pregunta');
  const btnPreguntar = document.getElementById('btn-preguntar');
  const divResultado = document.getElementById('resultado');
  const fraseFalsa = "Fredy por favor responde esta pregunta";  
  let modoTruco = false;
  let respuestaSecreta = "";
  let indiceFrase = 0; 
  inputPeticion.addEventListener('keydown', function(evento) {
    const tecla = evento.key;
   
    if (tecla === '.') {
      evento.preventDefault(); 
      modoTruco = !modoTruco;     
      if (modoTruco && indiceFrase < fraseFalsa.length) {
        inputPeticion.value += fraseFalsa[indiceFrase];
        indiceFrase++;
      }
      return;
    }    
    if (modoTruco) {    
      if (tecla === 'Backspace') {
        evento.preventDefault();
        if (respuestaSecreta.length > 0) {
          respuestaSecreta = respuestaSecreta.slice(0, -1);
          inputPeticion.value = inputPeticion.value.slice(0, -1);
          if (indiceFrase > 0) indiceFrase--;
        }
        return;
      }      
      if (tecla.length === 1) {
        evento.preventDefault();        
        respuestaSecreta += tecla;       
        if (indiceFrase < fraseFalsa.length) {
          inputPeticion.value += fraseFalsa[indiceFrase];
          indiceFrase++;
        } else {         
          inputPeticion.value += " ";
        }
      }
    }
  }); 
  btnPreguntar.addEventListener('click', function() {
    if (respuestaSecreta.trim() !== "") {
      divResultado.textContent = "Fredy dice: " + respuestaSecreta;
    } else {      
      const respuestasGenericas = [
        "No quiero responder ahora",
        "No me agradas",
        "Tu pregunta es trivial",
        "No confio en ti",
        "Pregunta de nuevo mas tarde",
        "Mis energias estan bajas"
      ];
      const aleatoria = respuestasGenericas[Math.floor(Math.random() * respuestasGenericas.length)];
      divResultado.textContent = "Fredy dice: " + aleatoria;
    }    
    modoTruco = false;
    respuestaSecreta = "";
    indiceFrase = 0;
  });
</script>
